💻 5-Line Integration Guide

Integrate NairaTag resolution into your wallet or DEX without forcing network switching:

With @nairatag/viem

import { createNairaTagResolver, classifyRecipientInput } from '@nairatag/viem';

// 1. Initialize resolver (Resolution RPC != Transaction RPC)
const nairatag = createNairaTagResolver({
  rpcUrl: 'http://127.0.0.1:9545' // or https://rpc.nairatag.org
});

// 2. Classify & Resolve in your Send flow
if (classifyRecipientInput(recipient) === 'nairatag') {
  const targetAddress = await nairatag.resolveName(recipient);
  // Returns checksummed address: 0x1111111111111111111111111111111111111111
}

With @nairatag/wagmi (React Hook)

import { useNairaTagAddress } from '@nairatag/wagmi';

function TransferComponent({ recipientInput }) {
  const { data: targetAddress, isLoading } = useNairaTagAddress({
    name: recipientInput,
    enabled: recipientInput.startsWith('₦')
  });

  return <span>Target: {targetAddress}</span>;
}
